<script>
const chat = document.getElementById("chat");
const form = document.getElementById("form");
const promptInput = document.getElementById("prompt");
const historyBox = document.getElementById("history");
const welcome = document.getElementById("welcome");
const sendButton = document.getElementById("send");

let conversations = [];
let currentId = null;

function createConversation() {
    const conversation = {
        id: Date.now() + Math.random(),
        title: "New chat",
        messages: []
    };

    conversations.unshift(conversation);
    currentId = conversation.id;
    renderHistory();
    renderChat();
}

function getCurrent() {
    return conversations.find(c => c.id === currentId);
}

function renderHistory() {
    historyBox.replaceChildren();

    conversations.forEach(c => {
        const item = document.createElement("div");
        item.className = "history-item" +
            (c.id === currentId ? " active" : "");
        item.textContent = c.title;
        item.title = c.title;

        item.onclick = () => {
            currentId = c.id;
            renderHistory();
            renderChat();
        };

        historyBox.appendChild(item);
    });
}

function renderChat() {
    chat.replaceChildren();

    const conversation = getCurrent();

    if (!conversation || conversation.messages.length === 0) {
        const box = welcome.cloneNode(true);
        box.style.display = "block";
        chat.appendChild(box);
        return;
    }

    conversation.messages.forEach(message => {
        addMessageElement(message.role, message.content);
    });
}

function addMessageElement(role, content) {
    const div = document.createElement("div");
    div.className = "message " + role;

    const label = document.createElement("div");
    label.className = "role";
    label.textContent = role === "user" ? "You" : "My AI";

    const text = document.createElement("div");
    text.textContent = content;

    div.append(label, text);
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
}

function makeReply(text) {
    const q = text.toLowerCase();

    if (q.includes("hello") || q.includes("hi")) {
        return "Hello! 👋 Welcome to My AI. How can I help you today?";
    }

    if (q.includes("your name")) {
        return "I am My AI, your demo virtual assistant.";
    }

    if (q.includes("python")) {
        return "Python is a popular programming language used in AI, data science, automation, and web development.\n\nWhat would you like to learn about Python?";
    }

    if (q.includes("html")) {
        return "HTML is used to structure webpages. CSS styles the page, and JavaScript makes it interactive.";
    }

    if (q.includes("math") || q.includes("calculate")) {
        return "I can help with maths in the connected AI version. This demo does not have a real AI reasoning engine yet.";
    }

    return "Thanks for your message! This is a demo chatbot, so I cannot generate a genuine AI answer yet.\n\nConnect a secure AI backend to enable real answers.";
}

form.addEventListener("submit", async (event) => {
    event.preventDefault();

    const text = promptInput.value.trim();
    if (!text || sendButton.disabled) return;

    if (!currentId) createConversation();

    const conversation = getCurrent();

    conversation.messages.push({
        role: "user",
        content: text
    });

    if (conversation.title === "New chat") {
        conversation.title = text.slice(0, 32);
    }

    promptInput.value = "";
    promptInput.style.height = "auto";
    renderHistory();
    renderChat();

    sendButton.disabled = true;

    const loading = document.createElement("div");
    loading.className = "message ai";
    loading.textContent = "Thinking...";
    chat.appendChild(loading);

    // Demo response. Replace with a backend API request
    // to connect a real AI model.
    await new Promise(resolve => setTimeout(resolve, 600));

    const reply = makeReply(text);

    conversation.messages.push({
        role: "assistant",
        content: reply
    });

    sendButton.disabled = false;
    renderChat();
    promptInput.focus();
});

promptInput.addEventListener("input", () => {
    promptInput.style.height = "auto";
    promptInput.style.height =
        Math.min(promptInput.scrollHeight, 140) + "px";
});

promptInput.addEventListener("keydown", event => {
    if (event.key === "Enter" && !event.shiftKey) {
        event.preventDefault();
        form.requestSubmit();
    }
});

document.getElementById("newChat").addEventListener("click", () => {
    createConversation();
    promptInput.focus();
});

createConversation();
</script>
